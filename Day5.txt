3. Find Maximum and Minimum in a List
------------------------------------------------

Program to find largest and smallest number.

def findMaxMin(arr):
    max_val = arr[0]
    min_val = arr[0]

    for num in arr:
        if num > max_val:
            max_val = num
        if num < min_val:
            min_val = num

    print("Largest value:", max_val)
    print("Smallest value:", min_val)


n = int(input("Enter range: "))
arr = []

for i in range(n):
    value = int(input("Enter number: "))
    arr.append(value)

findMaxMin(arr)

5. Remove Duplicate Elements from a List
------------------------------------------------

def removeDuplicates(arr):
    new_list = []
    for item in arr:
        if item not in new_list:
            new_list.append(item)
    return new_list

n = int(input("Enter range: "))
arr = []

for i in range(n):
    value = int(input("Enter number: "))
    arr.append(value)

print(removeDuplicates(arr))

6. Pattern Using While Loop
------------------------------------------------

i = 1
j = 10

while i < j:

    if i == 3 and j == 8:
        i += 1
        j -= 1
        continue

    print(i, "\t", j)

    i += 1
    j -= 1

7. Alternate Positive and Negative Numbers
------------------------------------------------

data = [1, -2, 3, -4, -1, 4]

positive = []
negative = []

# separate numbers
for val in data:
    if val >= 0:
        positive.append(val)
    else:
        negative.append(val)

final_list = []
i = j = 0

while i < len(positive) and j < len(negative):
    final_list.append(positive[i])
    final_list.append(negative[j])
    i += 1
    j += 1

final_list.extend(positive[i:])
final_list.extend(negative[j:])

print(final_list)
